/**
 * Optional public NASA endpoint fetch with strict timeout and offline fallback.
 * Never blocks gameplay or throws unhandled rejections.
 */
export async function fetchMissionReferenceData() {
  if (!API_CONFIG.enableLiveFetch) {
    return { sourceMode: 'Offline sample', data: DATA, livePayload: null };
  }

  const controller = new AbortController();
  const timeoutId = setTimeout(() => controller.abort(), API_CONFIG.timeoutMs);

  try {
    const response = await fetch(API_CONFIG.endpoint, { signal: controller.signal });
    clearTimeout(timeoutId);
    if (!response.ok) throw new Error(`HTTP ${response.status}`);
    const payload = await response.json();
    return { sourceMode: 'Live NASA data', data: DATA, livePayload: payload };
  } catch {
    clearTimeout(timeoutId);
    return { sourceMode: 'Offline sample', data: DATA, livePayload: null };
  }
}

/** Planet name without a leading "the " (CONTENT.planetInfo holds the display name). */
export const getPlanetName = id => CONTENT.planetInfo[id].name.replace(/^the /, '');

const getTopGate = () => GATES.filter(g => career.un.includes(g.id)).pop();
export const getCareerLevel = () => (getTopGate() ? getTopGate().lv : 0);
export const getCareerRank = () => (getTopGate() ? getTopGate().title : 'Launch Pad Rookie');

export const getCareerButtonLabel = () => `Career: Level ${getCareerLevel()} - ${getCareerRank()}`;

const countArrivals = (id, good) => (good ? career.good : career.arr)[id] || 0;

/** Human-readable requirement for one gate, with progress, e.g. "2 safe arrivals at Venus (1/2)". */
export const describeGateNeed = gate =>
  gate.need.map(([id, count, good]) => {
    const kind = good ? 'good' : 'safe';
    const plural = count > 1 ? 's' : '';
    const suffix = good ? ' (science above threshold)' : '';
    return `${count} ${kind} arrival${plural} at ${getPlanetName(id)}${suffix} (${Math.min(count, countArrivals(id, good))}/${count})`;
  }).join(', ');

export const getLockInfo = contract => {
  const gate = GATES.find(g => g.id === contract.id);
  return gate ? { level: gate.lv, need: describeGateNeed(gate) } : null;
};

/**
 * Records one finished mission, unlocks whatever it earned and returns the news to show in the debrief:
 * [{ k: 'lv', lv, title, planet } | { k: 'ach', n, t }].
 */
export function recordMission({ out, C, d, st, rtg, sci, hard }) {
  const ok = out !== 'lost';
  const news = [];
  const wasLost = career.streak > 0;

  if (ok) {
    career.arr[C.id] = (career.arr[C.id] || 0) + 1;
    career.tot++;
    career.streak = 0;
    if ((sci || 0) >= C.thr) career.good[C.id] = (career.good[C.id] || 0) + 1;
  } else {
    career.lost++;
    career.streak++;
  }

  for (let changed = true; changed;) {
    changed = false;
    GATES.forEach(gate => {
      const met = gate.need.every(([id, count, good]) => countArrivals(id, good) >= count);
      if (!career.un.includes(gate.id) && met) {
        career.un.push(gate.id);
        news.push({ k: 'lv', lv: gate.lv, title: gate.title, planet: getPlanetName(gate.id) });
        changed = true;
      }
    });
  }

  const context = { ok, out, C, d, st, rtg, wasLost, hard: !!hard };
  ACHIEVEMENTS.forEach(a => {
    if (!career.ach[a.id] && a.test(context)) {
      career.ach[a.id] = 1;
      news.push({ k: 'ach', n: a.n, t: a.t });
    }
  });

  saveCareer();
  return news;
}

export function resetCareer() {
  career = emptyCareer();
  saveCareer();
}

/**
 * Deterministic Mulberry32 PRNG (Rule 8: Same inputs + same seed = same output).
 * COSMETIC USE ONLY (e.g. jitter in the launch visuals). It must never decide win or loss:
 * flight outcomes are fully decided by resolveFlight() from the design's numbers.
 */
export function createSeededRng(seed = DEFAULT_SEED) {
  let state = seed >>> 0;
  return function nextRandom() {
    state += 0x6d2b79f5;
    let t = Math.imul(state ^ (state >>> 15), 1 | state);
    t ^= t + Math.imul(t ^ (t >>> 7), 61 | t);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

/* ---- Pure Geometry & Node Graph Helpers ---- */

export const findPartByUid = uid => appState.mission.parts.find(p => p.u === uid);
export const getPartSize = part => PART_DIMENSIONS[getDimensionKey(part.id)];
export const isStackNode = nodeName => ['top', 'bottom', 'base'].includes(nodeName);

export function getNodeOffset(part, nodeName) {
  const [width, height] = getPartSize(part);
  const node = ATTACHMENT_NODES[getNodeCategory(part.id)].find(n => n[0] === nodeName);
  let x = (node[1] - 0.5) * width;
  let y = (node[2] - 0.5) * height;
  if (part.flip) x = -x;

  const rad = (part.rot * Math.PI) / 180;
  const cos = Math.cos(rad);
  const sin = Math.sin(rad);
  return [x * cos - y * sin, x * sin + y * cos];
}

export function getNodeWorldPosition(part, nodeName) {
  const [offsetX, offsetY] = getNodeOffset(part, nodeName);
  return [part.x + offsetX, part.y + offsetY];
}

export function getSubtreeUids(rootUid) {
  const uids = [rootUid];
  for (let i = 0; i < uids.length; i++) {
    appState.mission.parts.forEach(part => {
      if (part.par === uids[i]) uids.push(part.u);
    });
  }
  return uids;
}

export function getOccupiedNodes() {
  const occupied = new Set();
  appState.mission.parts.forEach(part => {
    if (part.par != null) {
      occupied.add(`${part.par}:${part.pn}`);
      occupied.add(`${part.u}:${part.cn}`);
    }
  });
  return occupied;
}

export function getConnectedPartUids() {
  const connected = new Set();
  appState.mission.parts.forEach(part => {
    let current = part;
    let depth = 0;
    while (current && current.par != null && depth++ < 80) {
      current = findPartByUid(current.par);
    }
    if (current && getNodeCategory(current.id) === 'bus') {
      connected.add(part.u);
    }
  });
  return connected;
}

/* ---- Spacecraft Assembly Operations ---- */

export function createPartInstance(partId, x, y) {
  const part = {
    u: appState.mission.nu++,
    id: partId,
    x,
    y,
    rot: 0,
    flip: 0,
    par: null
  };
  appState.mission.parts.push(part);
  return part;
}

export function movePartSubtree(rootUid, deltaX, deltaY) {
  getSubtreeUids(rootUid).forEach(uid => {
    const part = findPartByUid(uid);
    part.x += deltaX;
    part.y += deltaY;
  });
}

export function deletePartSubtree(rootUid) {
  const toDelete = new Set(getSubtreeUids(rootUid));
  appState.mission.parts = appState.mission.parts.filter(part => !toDelete.has(part.u));
}

export function attachParts(child, childNode, host, hostNode) {
  const hasChildren = appState.mission.parts.some(p => p.par === child.u);
  if (!isStackNode(childNode) && !isStackNode(hostNode) && !hasChildren) {
    child.rot = 0;
    child.flip = hostNode.startsWith('L') ? 1 : 0;
  }

  const [hostX, hostY] = getNodeWorldPosition(host, hostNode);
  const [childOffsetX, childOffsetY] = getNodeOffset(child, childNode);
  movePartSubtree(child.u, hostX - childOffsetX - child.x, hostY - childOffsetY - child.y);

  child.par = host.u;
  child.cn = childNode;
  child.pn = hostNode;
}

export function mirrorPartIfEnabled(childPart) {
  const host = findPartByUid(childPart.par);
  if (
    !appState.mission.mirror ||
    isStackNode(childPart.pn) ||
    host.rot !== 0 ||
    DATA.parts[childPart.id].g === 'sci' ||
    appState.mission.parts.length >= MAX_PARTS
  ) {
    return;
  }

  const oppositeSide = childPart.pn.startsWith('L') ? 'R' : 'L';
  const oppositeNode = oppositeSide + childPart.pn.slice(1);
  if (getOccupiedNodes().has(`${host.u}:${oppositeNode}`)) return;

  const twinPart = createPartInstance(childPart.id, 0, 0);
  attachParts(twinPart, childPart.cn, host, oppositeNode);
}

export function autoAttachPart(partId) {
  if (!canCurrentRoleEditPart(partId)) {
    return { ok: false, message: 'That part belongs to another role. Pass the keyboard.' };
  }

  const category = getNodeCategory(partId);
  const connected = getConnectedPartUids();
  const hosts = appState.mission.parts
    .filter(p => connected.has(p.u))
    .sort((a, b) => a.y - b.y);
  const bus = appState.mission.parts.find(p => getNodeCategory(p.id) === 'bus');

  if (category === 'bus') {
    if (bus) {
      bus.id = partId;
    } else {
      saveUndoSnapshot();
      createPartInstance(partId, 400, 170);
    }
    return { ok: true };
  }

  if (!bus) {
    return { ok: false, message: 'Start with a spacecraft bus: it is the root everything bolts onto.' };
  }
  if (appState.mission.parts.length >= MAX_PARTS) {
    return { ok: false, message: `Part limit reached (${MAX_PARTS}).` };
  }

  const occupied = getOccupiedNodes();
  const isNodeFree = (host, node) => !occupied.has(`${host.u}:${node}`);
  const hasNode = (host, node) => ATTACHMENT_NODES[getNodeCategory(host.id)].some(n => n[0] === node);

  saveUndoSnapshot();

  if (category === 'eng') {
    const existingEngine = appState.mission.parts.find(p => getNodeCategory(p.id) === 'eng');
    if (existingEngine) {
      existingEngine.id = partId;
      if (existingEngine.par != null) {
        attachParts(existingEngine, 'top', findPartByUid(existingEngine.par), existingEngine.pn);
      }
      return { ok: true };
    }
    const bottomHost = [...hosts].reverse().find(h => hasNode(h, 'bottom') && isNodeFree(h, 'bottom'));
    if (!bottomHost) return { ok: false, message: 'Nothing to bolt an engine under.' };
    attachParts(createPartInstance(partId, 0, 0), 'top', bottomHost, 'bottom');
    return { ok: true };
  }

  if (category === 'tank') {
    const lowestStack = [...hosts].reverse().find(h => ['bus', 'tank'].includes(getNodeCategory(h.id)));
    const attachedBottom = appState.mission.parts.find(p => p.par === lowestStack.u && p.pn === 'bottom');
    const newTank = createPartInstance('tank', 0, 0);

    if (attachedBottom) {
      attachedBottom.par = null;
      attachParts(newTank, 'top', lowestStack, 'bottom');
      attachParts(attachedBottom, attachedBottom.cn, newTank, 'bottom');
    } else {
      attachParts(newTank, 'top', lowestStack, 'bottom');
    }
    return { ok: true };
  }

  if (category === 'ant') {
    const topHost = hosts.find(h => ['bus', 'tank'].includes(getNodeCategory(h.id)) && isNodeFree(h, 'top'));
    if (!topHost) return { ok: false, message: 'No free mounting point on top. Move something first.' };
    attachParts(createPartInstance(partId, 0, 0), 'base', topHost, 'top');
    return { ok: true };
  }

  for (const host of hosts) {
    for (const sideNode of ['R1', 'L1', 'R2', 'L2']) {
      if (hasNode(host, sideNode) && isNodeFree(host, sideNode)) {
        const part = createPartInstance(partId, 0, 0);
        attachParts(part, 'in', host, sideNode);
        mirrorPartIfEnabled(part);
        return { ok: true };
      }
    }
  }

  return { ok: false, message: 'No free mounting point. Drag a part off the ship, or add a tank for more mounts.' };
}

export function calculateTelemetry(missionInput = appState.mission) {
  const threshold = missionInput.hard ? DATA.marginPolicies.hard.v : DATA.marginPolicies.easy.v;
  appState.marginThreshold = threshold;

  const contract = DATA.contracts[missionInput.ci];
  const orbit = contract.orbits[missionInput.oi];
  const launcher = DATA.launchers[missionInput.L];
  const tankSpec = DATA.tank;

  const connectedUids = getConnectedPartUids();
  const connectedParts = missionInput.parts.filter(p => connectedUids.has(p.u));

  const busPart = connectedParts.find(p => getNodeCategory(p.id) === 'bus');
  const engineParts = connectedParts.filter(p => getNodeCategory(p.id) === 'eng');

  const busSpec = DATA.parts[busPart ? busPart.id : 'bus_std'];
  const engineSpec = DATA.engines[engineParts[0] ? engineParts[0].id : 'chem'];

  const telemetry = {
    C: contract,
    O: orbit,
    B: busSpec,
    E: engineSpec,
    L: launcher,
    dry: busSpec.m + engineParts.length * engineSpec.m,
    load: busSpec.pw,
    gen: 0,
    rg: 0, // radioisotope generator power: does not depend on sunlight
    wh: 0,
    mb: 0,
    gb: DATA.constants.baseStorageGb.value,
    rec: 0,
    data: 0,
    sci: 0,
    tk: 0,
    cost: busSpec.c + engineParts.length * engineSpec.c + launcher.c
  };

  let propellantMass = 0;
  let weightedX = 0;
  let weightedY = 0;
  let totalMass = 0;
  const seenInstruments = {};

  connectedParts.forEach(part => {
    let partMass = 0;
    const category = getNodeCategory(part.id);

    if (part.id === 'tank') {
      telemetry.tk++;
      telemetry.dry += tankSpec.dry;
      telemetry.cost += tankSpec.c;
      propellantMass += tankSpec.prop;
      partMass = tankSpec.dry + tankSpec.prop;
    } else if (category === 'eng') {
      partMass = engineSpec.m;
    } else if (category === 'bus') {
      partMass = busSpec.m;
    } else {
      const spec = DATA.parts[part.id];
      partMass = spec.m;
      telemetry.dry += spec.m;
      telemetry.cost += spec.c;
      telemetry.load += spec.pw || 0;
      if (spec.rtg) telemetry.rg += spec.gen || 0;
      else telemetry.gen += spec.gen || 0;
      telemetry.wh += spec.wh || 0;
      telemetry.mb = Math.max(telemetry.mb, spec.mb || 0);

      if (spec.gb) {
        telemetry.gb += spec.gb;
        telemetry.rec++;
      }
      telemetry.data += spec.data || 0;
      if (spec.sci) {
        telemetry.sci += spec.sci * (seenInstruments[part.id] ? 0.25 : 1);
        seenInstruments[part.id] = true;
      }
    }

    totalMass += partMass;
    weightedX += partMass * part.x;
    weightedY += partMass * part.y;
  });

  telemetry.comx = totalMass ? weightedX / totalMass : 0;
  telemetry.comy = totalMass ? weightedY / totalMass : 0;
  telemetry.engx = engineParts[0] ? engineParts[0].x : telemetry.comx;
  telemetry.off = engineParts.length ? Math.abs(telemetry.comx - telemetry.engx) : 0;
  telemetry.box = connectedParts.length
    ? [Math.min(...connectedParts.map(p => p.y)) - 30, Math.max(...connectedParts.map(p => p.y)) + 80]
    : [0, 0];

  telemetry.m0 = telemetry.dry + propellantMass;
  telemetry.dv = engineSpec.isp * DATA.constants.g0.value * Math.log(telemetry.m0 / telemetry.dry);
  telemetry.sun = telemetry.gen * DATA.solarFluxFactor[contract.dest].value * (1 - orbit.ecl) + telemetry.rg;
  telemetry.ecH = orbit.ecl * orbit.per;
  telemetry.down = telemetry.mb * DATA.commsDistanceFactor[contract.dest].value * orbit.contact * 86.4;
  telemetry.days = Math.max(
    DATA.transitDaysBase[contract.dest].value * engineSpec.tf,
    engineSpec.dvDay ? (orbit.dv * 1000) / engineSpec.dvDay : 0
  );
  // Far destinations carry their own design life (the trip alone can take years). Harsh radiation (rad 2) wears even hardened buses.
  telemetry.baseLife = contract.life || DATA.constants.designLifeMonths.value;
  telemetry.life = telemetry.baseLife * (orbit.rad && !busSpec.hard ? 0.6 : orbit.rad > 1 ? 0.7 : 1) - telemetry.days / 30;

  // Section 6.2 margin convention: margin = (capacity - predictedUse) / predictedUse
  const buildCheckRow = (name, category, used, capacity, unit, src, verified) => {
    const margin = used <= 0 ? 1 : capacity > 0 ? (capacity - used) / used : -1;
    const lesson = CONTENT.lessons[name];
    return {
      id: name.toLowerCase().replace(/\s+/g, '_'),
      category,
      n: name,
      use: used,
      cap: capacity,
      u: unit,
      mg: margin,
      pass: margin >= 0,
      severity: margin < 0 ? 'critical' : margin < threshold ? 'warning' : 'info',
      why: lesson.why,
      fix: lesson.fix,
      src,
      verified
    };
  };

  const requiredContactHours = telemetry.data
    ? (telemetry.mb ? telemetry.data / (telemetry.mb * DATA.commsDistanceFactor[contract.dest].value * 3.6) : 99)
    : 0;

  telemetry.rows = [
    buildCheckRow('Mass', 'mass', telemetry.m0, launcher.cap, 'kg', launcher.src, launcher.verified),
    buildCheckRow('Delta-v', 'launch', orbit.dv * 1000, telemetry.dv, 'm/s', orbit.src, orbit.verified),
    buildCheckRow('Power', 'power', telemetry.load, telemetry.sun, 'W', DATA.solarFluxFactor[contract.dest].src, false),
    buildCheckRow('Battery', 'power', telemetry.load * telemetry.ecH, telemetry.wh * DATA.constants.batteryDepthOfDischarge.value, 'Wh', DATA.constants.batteryDepthOfDischarge.src, false),
    buildCheckRow('Budget', 'budget', telemetry.cost, contract.cap, '$M', contract.src, contract.verified),
    buildCheckRow('Data storage', 'data', telemetry.data * (1 - orbit.contact), telemetry.gb, 'Gb', DATA.constants.baseStorageGb.src, false),
    buildCheckRow('Comms', 'comms', requiredContactHours, orbit.contact * 24, 'h/day', DATA.commsDistanceFactor[contract.dest].src, false),
    buildCheckRow('Time', 'time', contract.months, telemetry.life, 'months', DATA.constants.designLifeMonths.src, false)
  ];

  if (engineSpec.pw) {
    telemetry.rows.push(
      buildCheckRow('Ion power', 'power', engineSpec.pw, telemetry.gen + telemetry.rg, 'W', engineSpec.src, engineSpec.verified)
    );
  }

  const getRow = name => telemetry.rows.find(r => r.n === name);
  telemetry.g = getRow;

  const flightModel = DATA.flightModel;
  const floatingCount = missionInput.parts.length - connectedParts.length;
  const isEngineValid = busPart && engineParts.length === 1 && engineParts[0].rot === 0 && engineParts[0].y > busPart.y;

  telemetry.checks = [
    {
      t: 'A spacecraft bus is the root of the ship',
      st: busPart ? 2 : 0,
      why: 'Everything must bolt onto the bus.',
      fix: 'Add a spacecraft bus from the Structure list.'
    },
    {
      t: 'Exactly one engine, nozzle down, below the bus',
      st: isEngineValid ? 2 : 0,
      why: 'Rockets must apply thrust axially from the aft stack.',
      fix: 'Attach one engine facing down beneath your fuel tanks.'
    },
    {
      t: 'Every part is attached',
      st: floatingCount ? 0 : 2,
      why: `${floatingCount} part(s) are unattached (highlighted red).`,
      fix: 'Drag unattached parts onto a glowing attachment node or delete them.'
    },
    {
      t: 'Balanced: thrust line through the center of mass',
      st: telemetry.off <= flightModel.balanceGoodPx ? 2 : telemetry.off <= flightModel.balanceLimitPx ? 1 : 0,
      why: `Center of mass is ${telemetry.off.toFixed(0)} px off the thrust line (tumbles above ${flightModel.balanceLimitPx} px).`,
      fix: 'Mirror heavy parts symmetrically left and right of the centerline.'
    }
  ];

  const dataReturnRatio = telemetry.data
    ? Math.min(1, telemetry.down / telemetry.data) * (getRow('Data storage').mg < 0 ? telemetry.gb / (telemetry.data * (1 - orbit.contact)) : 1)
    : 0;

  telemetry.ret = dataReturnRatio;
  telemetry.sciF =
    telemetry.sci *
    orbit.res *
    dataReturnRatio *
    (Math.max(0, telemetry.life) / telemetry.baseLife) *
    (getRow('Power').mg < 0 ? 0.5 : 1) *
    (getRow('Battery').mg < 0 ? 0.6 : 1);

  const hasHealthyMargin = name => getRow(name).mg >= threshold;
  const isPositiveMargin = name => getRow(name).mg >= 0;

  telemetry.sign = {
    Director: isPositiveMargin('Mass') && isPositiveMargin('Budget') && isPositiveMargin('Time'),
    Engineer:
      hasHealthyMargin('Delta-v') &&
      hasHealthyMargin('Power') &&
      hasHealthyMargin('Battery') &&
      (!engineSpec.pw || hasHealthyMargin('Ion power')) &&
      telemetry.checks[3].st === 2,
    Science: telemetry.sciF >= contract.target,
    Comms: hasHealthyMargin('Comms') && hasHealthyMargin('Data storage') && telemetry.data > 0
  };

  telemetry.revItems = [
    ...telemetry.rows.map(r => [r.n, r.mg < 0 ? 0 : r.mg < threshold ? 1 : 2]),
    ...telemetry.checks.map(c => [c.t, c.st]),
    ['Science vs threshold goal', telemetry.sciF >= contract.target ? 2 : telemetry.sciF >= contract.thr ? 1 : 0]
  ];

  const itemCount = telemetry.revItems.length;
  const countStatus = code => telemetry.revItems.filter(item => item[1] === code).length;
  telemetry.rev = {
    n: itemCount,
    g: countStatus(2),
    y: countStatus(1),
    r: countStatus(0),
    pct: countStatus(2) / itemCount,
    pass: countStatus(2) / itemCount >= DATA.reviewThreshold.v
  };

  // Launch reliability: a visible number that scales the final score. It is never rolled against.
  const marginPenalty = telemetry.rows.reduce(
    (sum, r) => (r.mg >= 0 && r.mg < threshold ? sum + (threshold - r.mg) * flightModel.amberPenaltyFactor : sum),
    0
  );
  const balancePenalty = Math.min(
    flightModel.balancePenaltyAtLimit,
    (telemetry.off / flightModel.balanceLimitPx) * flightModel.balancePenaltyAtLimit
  );
  const signOffBonus = telemetry.signed ? flightModel.signOffBonus : 0;
  const reviewPenalty = telemetry.rev.pass ? 0 : flightModel.reviewFailPenalty;
  const baseReliability = launcher.rel * busSpec.rel;
  const adjustment = -marginPenalty - balancePenalty + signOffBonus - reviewPenalty;

  telemetry.reliabilityParts = { launcher: launcher.rel, bus: busSpec.rel, adjustment };
  telemetry.reliability = Math.max(
    flightModel.reliabilityMin,
    Math.min(flightModel.reliabilityMax, baseReliability + adjustment)
  );
  telemetry.bad = telemetry.checks.slice(0, 3).find(c => c.st === 0);

  return telemetry;
}

/**
 * Fills {placeholders} in a CONTENT template. Numbers are rounded for display.
 * A key ending in "Pct" is a fraction shown as a whole percent (0.94 -> 94).
 */
export function fillTemplate(template, values = {}) {
  return template.replace(/\{(\w+)\}/g, (match, key) => {
    const value = values[key];
    if (typeof value !== 'number') return value === undefined ? match : String(value);
    return key.endsWith('Pct') ? String(Math.round(value * 100)) : Math.round(value).toLocaleString();
  });
}

function doesEventTrigger(event, telemetry) {
  const { trigger } = event;
  if (trigger.dest !== undefined) return telemetry.C.dest === trigger.dest;
  const value = telemetry.O[trigger.orbitField];
  if (trigger.gte !== undefined && value < trigger.gte) return false;
  if (trigger.lte !== undefined && value > trigger.lte) return false;
  return true;
}

function isEventImmune(event, telemetry) {
  if (event.imm === 'hard') return !!telemetry.B.hard;
  if (event.imm === 'rec') return telemetry.rec > 0;
  return !!telemetry.B.red;
}

/**
 * Every event this contract and orbit will cause, and whether the design is immune.
 * Same design = same list, every time.
 */
export function getTriggeredEvents(telemetry) {
  return DATA.events
    .filter(event => doesEventTrigger(event, telemetry))
    .map(event => ({
      id: event.id,
      immune: isEventImmune(event, telemetry),
      scienceFactor: isEventImmune(event, telemetry) ? 1 : event.scienceFactor
    }));
}

function scoreFlight(telemetry, outcome, science) {
  const weights = DATA.scoringWeights;
  const contract = telemetry.C;

  let stars = outcome === 'lost' ? 0 : 1;
  if (science >= contract.target) stars = 2;
  if (stars === 2 && telemetry.signed && telemetry.cost <= contract.cap * weights.threeStarBudgetFraction) stars = 3;

  const score = outcome === 'lost'
    ? 0
    : Math.round(
        (science * weights.scienceMultiplier +
          (contract.cap - telemetry.cost) * weights.budgetSavingsMultiplier +
          stars * weights.starBonus) *
          (telemetry.signed ? weights.fullSignOffMultiplier : 1) *
          telemetry.reliability
      );

  return { stars, score };
}

/**
 * Resolves a whole flight from the design's numbers. Pure and deterministic.
 * The launch screen (text today, a visual simulation later) only PLAYS the returned
 * `steps`; it never decides the outcome.
 *
 * steps: ordered [{ key, ok, fatal, values }]. `key` names a CONTENT.flightLog template,
 * `values` holds the raw numbers for it, `fatal` means the mission ends on that step.
 */
export function resolveFlight(telemetry) {
  const model = DATA.flightModel;
  const contract = telemetry.C;
  const steps = [];

  const finish = (outcome, science, events, reasons) => ({
    outcome,
    science,
    events,
    reasons,
    steps,
    reliability: telemetry.reliability,
    ...scoreFlight(telemetry, outcome, science)
  });

  const lose = (key, values, reasonKey) => {
    steps.push({ key, ok: false, fatal: true, values });
    const reason = CONTENT.flightFailures[reasonKey];
    return finish('lost', 0, [], [`${reason.why} Fix: ${reason.fix}`]);
  };

  const deltaV = telemetry.g('Delta-v');

  steps.push({
    key: 'liftoff',
    ok: true,
    values: { launcher: telemetry.L.n, massKg: telemetry.m0, reliabilityPct: telemetry.reliability }
  });

  if (telemetry.off > model.balanceLimitPx) {
    return lose('tumble', { offPx: telemetry.off, limitPx: model.balanceLimitPx }, 'tumble');
  }
  steps.push({ key: 'ascent', ok: true, values: { reliabilityPct: telemetry.reliability } });

  const feasibility = getFeasibility(contract);
  steps.push({
    key: 'feasibility',
    ok: true,
    values: {
      name: contract.n,
      cls: feasibility.cls,
      label: feasibility.lab,
      notes: feasibility.notes.join('; '),
      haveMs: telemetry.dv,
      needMs: deltaV.use,
      haveW: telemetry.sun,
      needW: telemetry.load,
      days: telemetry.days
    }
  });

  if (contract.doom) {
    // The Sun cannot be reached by any design, so this branch never depends on the numbers above.
    CONTENT.sunDive.lines.forEach(([text, cls]) => {
      steps.push({ key: 'sun_dive', ok: cls !== 'bad', values: { text } });
    });
    steps.push({ key: 'sun_boom', ok: false, fatal: true, values: { text: CONTENT.sunDive.boom } });
    const sunReasons = CONTENT.sunDive.why.map((line, index) =>
      index === 0 ? `${line} Your craft had ${Math.round(telemetry.dv).toLocaleString()} m/s.` : line
    );
    return finish('lost', 0, [], sunReasons);
  }

  steps.push({ key: 'transfer', ok: deltaV.mg >= 0, values: { needMs: deltaV.use, haveMs: telemetry.dv } });
  if (deltaV.mg < 0) {
    return lose('deltav_short', { shortMs: deltaV.use - telemetry.dv }, 'deltav');
  }
  if (telemetry.E.pw && telemetry.g('Ion power').mg < 0) {
    return lose('ion_short', { needW: telemetry.E.pw }, 'ion');
  }

  steps.push({ key: 'arrival', ok: true, values: { days: telemetry.days, orbit: telemetry.O.n.toLowerCase() } });
  steps.push({ key: 'briefing', ok: true, values: { contractId: contract.id } });

  const events = getTriggeredEvents(telemetry);
  events.forEach(event => {
    steps.push({ key: 'event', ok: event.immune, values: { eventId: event.id, scienceFactor: event.scienceFactor } });
  });

  const science = events.reduce((total, event) => total * event.scienceFactor, telemetry.sciF);
  steps.push({
    key: 'operations',
    ok: true,
    values: { returnedPct: telemetry.ret, science, target: contract.target }
  });

  const outcome = science >= contract.target ? 'success' : science > 0 ? 'partial' : 'lost';

  // Debrief reasons: every thin row, every lost-data note, every event that hurt, and the reliability maths.
  const reasons = telemetry.rows
    .filter(r => r.mg < appState.marginThreshold)
    .map(r => `${r.n}: ${r.why} Fix: ${r.fix} [${r.src}]`);

  if (telemetry.ret < 1 && telemetry.data) {
    reasons.push(fillTemplate(CONTENT.flightFailures.dataReturn, { returnedPct: telemetry.ret }));
  }

  events.filter(event => !event.immune).forEach(event => {
    const text = CONTENT.eventText[event.id];
    reasons.push(`${text.n}: ${text.hitT} ${fillTemplate(CONTENT.flightFailures.eventLoss, { lossPct: 1 - event.scienceFactor })} Fix: ${text.fix}`);
  });

  const reliabilityLine = fillTemplate(CONTENT.flightFailures.reliability, {
    reliabilityPct: telemetry.reliability,
    launcherPct: telemetry.reliabilityParts.launcher,
    busPct: telemetry.reliabilityParts.bus,
    adjustmentPct: telemetry.reliabilityParts.adjustment
  });
  reasons.push(reliabilityLine + (telemetry.reliabilityParts.adjustment < 0 ? CONTENT.flightFailures.reliabilityFix : ''));

  return finish(outcome, science, events, reasons);
}

/**
 * Best delta-v (km/s) the game lets you build: super-heavy launcher, hardened bus, as many tanks as fit.
 * Used only to label how hard a destination is before the player designs anything.
 */
const BEST_BUILDABLE_DV = (() => {
  const tank = DATA.tank;
  const bus = DATA.parts.bus_rad;
  const result = {};

  for (const key of ['chem', 'nuclear']) {
    const engine = DATA.engines[key];
    let best = 0;
    for (let tanks = 1; tanks <= 8; tanks++) {
      const dry = bus.m + engine.m + tanks * tank.dry + 250;
      const wet = dry + tanks * tank.prop;
      if (wet <= DATA.launchers.super.cap) {
        best = Math.max(best, engine.isp * DATA.constants.g0.value * Math.log(wet / dry));
      }
    }
    result[key] = best / 1000;
  }
  return result;
})();

/**
 * Rates a destination before design: Feasible, Hard, Extreme, Beyond current engines, or IMPOSSIBLE.
 * Returns { lab, cls, notes[], dv } where cls is a colour class (g / y / r).
 */
export function getFeasibility(contract) {
  if (contract.doom) {
    return { lab: 'IMPOSSIBLE', cls: 'r', notes: ['no spacecraft can survive it; try it and see why'], dv: Infinity };
  }

  const cheapestDv = Math.min(...contract.orbits.map(o => o.dv));
  const chemical = BEST_BUILDABLE_DV.chem;
  const nuclear = BEST_BUILDABLE_DV.nuclear;
  const sunlight = DATA.solarFluxFactor[contract.dest].value;
  const commsFactor = DATA.commsDistanceFactor[contract.dest].value;
  const tripDays = DATA.transitDaysBase[contract.dest].value;

  let lab;
  let cls;
  if (cheapestDv <= chemical * 0.65) { lab = 'Feasible'; cls = 'g'; }
  else if (cheapestDv <= chemical * 0.95) { lab = 'Hard: giant chemical rocket'; cls = 'y'; }
  else if (cheapestDv <= nuclear * 0.95) { lab = 'Extreme: needs nuclear engine'; cls = 'y'; }
  else { lab = 'Beyond current engines'; cls = 'r'; }

  const notes = [
    `needs at least ${cheapestDv.toFixed(1)} km/s of delta-v`,
    `trip about ${tripDays < 120 ? `${tripDays} days` : `${(tripDays / 365).toFixed(1)} years`} (chemical engine)`
  ];
  if (sunlight < 0.2) notes.push(`sunlight only ${(sunlight * 100).toFixed(1)}% of Earth’s: bring radioisotope generators`);
  else if (sunlight > 2.5) notes.push(`sunlight ${sunlight}x Earth’s: heat is the enemy`);
  if (commsFactor < 0.05) notes.push(`weak signal: only ${(commsFactor * 100).toFixed(1)}% of the data rate, choose light instruments`);

  return { lab, cls, notes, dv: cheapestDv };
}

export const AudioEngine = (() => {
  let ctx = null;
  let masterGain = null;
  let isSoundOn = false;

  const init = () => {
    if (ctx) return;
    const AudioCtx = window.AudioContext || window.webkitAudioContext;
    if (!AudioCtx) return;
    ctx = new AudioCtx();
    masterGain = ctx.createGain();
    masterGain.gain.value = isSoundOn ? 0.45 : 0;
    masterGain.connect(ctx.destination);
  };

  const playTone = (freq, delay, duration, type = 'sine', volume = 0.2, endFreq = null) => {
    if (!ctx || !isSoundOn) return;
    const osc = ctx.createOscillator();
    const gain = ctx.createGain();
    const startTime = ctx.currentTime + delay;

    osc.type = type;
    osc.frequency.setValueAtTime(freq, startTime);
    if (endFreq) osc.frequency.exponentialRampToValueAtTime(endFreq, startTime + duration);

    gain.gain.setValueAtTime(volume, startTime);
    gain.gain.exponentialRampToValueAtTime(0.001, startTime + duration);

    osc.connect(gain);
    gain.connect(masterGain);
    osc.start(startTime);
    osc.stop(startTime + duration);
  };

  return {
    init,
    isEnabled: () => isSoundOn,
    toggleMute() {
      init();
      isSoundOn = !isSoundOn;
      if (masterGain) masterGain.gain.value = isSoundOn ? 0.45 : 0;
      return isSoundOn;
    },
    click: () => playTone(660, 0, 0.05, 'square', 0.06),
    attach: () => {
      playTone(520, 0, 0.08, 'triangle', 0.14);
      playTone(780, 0.07, 0.1, 'triangle', 0.11);
    },
    remove: () => playTone(420, 0, 0.12, 'triangle', 0.12, 240),
    warn: () => {
      playTone(310, 0, 0.14, 'sawtooth', 0.07);
      playTone(310, 0.2, 0.14, 'sawtooth', 0.07);
    },
    tick: () => playTone(900, 0, 0.04, 'square', 0.05),
    launch: () => playTone(65, 0, 2.5, 'sawtooth', 0.14, 110),
    win: () => [523, 659, 784, 1047].forEach((f, i) => playTone(f, i * 0.14, 0.45, 'triangle', 0.16)),
    partial: () => [523, 659].forEach((f, i) => playTone(f, i * 0.2, 0.45, 'triangle', 0.14)),
    fail: () => playTone(120, 0, 1.0, 'sawtooth', 0.22, 35)
  };
})();

function rotateSelectedPart(direction) {

    q.x = pivot[0] + relX * cos - relY * sin;
    q.y = pivot[1] + relX * sin + relY * cos;
    q.rot = (q.rot + angle + 360) % 360;
  });

  AudioEngine.click();
  refreshBuilderView();
}

function flipSelectedPart() {
  const part = appState.selectedPartUid && findPartByUid(appState.selectedPartUid);
  if (!part || !canCurrentRoleEditPart(part.id)) return;

  saveUndoSnapshot();
  const pivotX = part.par != null ? getNodeWorldPosition(part, part.cn)[0] : part.x;

  getSubtreeUids(part.u).forEach(uid => {
    const q = findPartByUid(uid);
    q.x = 2 * pivotX - q.x;
    q.flip ^= 1;
    q.rot = (360 - q.rot) % 360;
  });

  AudioEngine.click();
  refreshBuilderView();
}

function deleteSelectedPart() {
  const part = appState.selectedPartUid && findPartByUid(appState.selectedPartUid);
  if (!part || !canCurrentRoleEditPart(part.id)) return;

  saveUndoSnapshot();
  deletePartSubtree(part.u);
  appState.selectedPartUid = null;
  AudioEngine.remove();
  refreshBuilderView();
}

function undoLastAction() {
  const snapshot = appState.undoHistory.pop();
  if (!snapshot) return showToast('Nothing to undo.');

  [appState.mission.parts, appState.mission.nu] = JSON.parse(snapshot);
  appState.selectedPartUid = null;
  refreshBuilderView();
}

export function bindGlobalControls() {
  document.addEventListener('pointerdown', e => {
    if (appState.mission.stage !== 2) return;

    const placedEl = e.target.closest('.pt[data-u]');
    const paletteEl = e.target.closest('[data-p]');
    if (!placedEl && !paletteEl) return;
    e.preventDefault();

    if (paletteEl) {
      appState.activeDrag = { nid: paletteEl.dataset.p, x: e.clientX, y: e.clientY, mv: false };
      return;
    }

    const part = findPartByUid(+placedEl.dataset.u);
    appState.selectedPartUid = part.u;
    if (!canCurrentRoleEditPart(part.id)) {
      showToast('That part belongs to another role. Pass the keyboard.');
      refreshBuilderView();
      return;
    }

    appState.activeDrag = {
      u: part.u,
      w0: clientToWorldCoords(e),
      x: e.clientX,
      y: e.clientY,
      mv: false
    };
    refreshBuilderView();
  });

  document.addEventListener('pointermove', e => {
    const drag = appState.activeDrag;
    if (!drag) return;

    if (!drag.mv) {
      if (Math.hypot(e.clientX - drag.x, e.clientY - drag.y) < 5) return;

      if (drag.nid) {
        const partId = drag.nid;
        const hasCategory = cat => appState.mission.parts.some(p => getNodeCategory(p.id) === cat);

        if (!canCurrentRoleEditPart(partId)) {
          appState.activeDrag = null;
          return showToast('That part belongs to another role. Pass the keyboard.');
        }
        if (hasCategory('bus') && getNodeCategory(partId) === 'bus') {
          appState.activeDrag = null;
          return showToast('One bus per ship. Click a different bus in the list to swap it.');
        }
        if (hasCategory('eng') && getNodeCategory(partId) === 'eng') {
          appState.activeDrag = null;
          return showToast('One main engine. Click a different engine in the list to swap it.');
        }
        if (appState.mission.parts.length >= MAX_PARTS) {
          appState.activeDrag = null;
          return showToast(`Part limit reached (${MAX_PARTS}).`);
        }

        saveUndoSnapshot();
        const worldPos = clientToWorldCoords(e);
        const newPart = createPartInstance(partId, worldPos[0], worldPos[1]);
        appState.selectedPartUid = newPart.u;
        appState.activeDrag = { u: newPart.u, w0: worldPos, x: e.clientX, y: e.clientY, mv: true };
      } else {
        saveUndoSnapshot();
        findPartByUid(drag.u).par = null;
        drag.mv = true;
      }

      appState.activeDrag.orig = {};
      getSubtreeUids(appState.activeDrag.u).forEach(uid => {
        const q = findPartByUid(uid);
        appState.activeDrag.orig[uid] = [q.x, q.y];
      });
      AudioEngine.click();
    }

    const activeDrag = appState.activeDrag;
    const worldPos = clientToWorldCoords(e);
    const dx = worldPos[0] - activeDrag.w0[0];
    const dy = worldPos[1] - activeDrag.w0[1];
    const draggedPart = findPartByUid(activeDrag.u);

    getSubtreeUids(activeDrag.u).forEach(uid => {
      const q = findPartByUid(uid);
      q.x = activeDrag.orig[uid][0] + dx;
      q.y = activeDrag.orig[uid][1] + dy;
    });

    activeDrag.sn = null;
    if (getNodeCategory(draggedPart.id) !== 'bus') {
      const occupied = getOccupiedNodes();
      const draggedTree = new Set(getSubtreeUids(activeDrag.u));
      let closestDist = SNAP_RADIUS_PX;

      const hasChildren = appState.mission.parts.some(q => q.par === draggedPart.u);
      const origFlip = draggedPart.flip;
      const origRot = draggedPart.rot;

      ATTACHMENT_NODES[getNodeCategory(draggedPart.id)].forEach(([childNode]) => {
        if (occupied.has(`${draggedPart.u}:${childNode}`)) return;

        appState.mission.parts.forEach(host => {
          if (draggedTree.has(host.u)) return;

          ATTACHMENT_NODES[getNodeCategory(host.id)].forEach(([hostNode]) => {
            if (occupied.has(`${host.u}:${hostNode}`) || isStackNode(childNode) !== isStackNode(hostNode)) return;

            if (!isStackNode(childNode) && !hasChildren) {
              draggedPart.rot = 0;
              draggedPart.flip = hostNode.startsWith('L') ? 1 : 0;
            }

            const a = getNodeWorldPosition(draggedPart, childNode);
            const b = getNodeWorldPosition(host, hostNode);
            const distance = Math.hypot(a[0] - b[0], a[1] - b[1]);

            draggedPart.flip = origFlip;
            draggedPart.rot = origRot;

            if (distance < closestDist) {
              closestDist = distance;
              activeDrag.sn = { h: host.u, hn: hostNode, cn: childNode };
            }
          });
        });
      });

      if (activeDrag.sn) {
        attachParts(draggedPart, activeDrag.sn.cn, findPartByUid(activeDrag.sn.h), activeDrag.sn.hn);
        draggedPart.par = null;
      }
    }

    refreshBuilderView();
  });

  document.addEventListener('pointerup', e => {

    if (!drag) return;
    appState.activeDrag = null;

    }

    if (appState.mission.stage !== 2 || /SELECT|TEXTAREA|INPUT/.test(e.target.tagName)) return;
    const key = e.key.toLowerCase();

    if (e.ctrlKey || e.metaKey) {
      if (key === 'z') {
        e.preventDefault();
        undoLastAction();
      }
      return;
    }

    if (key === 'r') rotateSelectedPart(e.shiftKey ? -1 : 1);
    else if (key === 'f') flipSelectedPart();
    else if (key === 'delete' || key === 'backspace') {

    e.target.textContent = `Sound: ${isNowOn ? 'on' : 'off'}`;
  });


  $('#copy').addEventListener('click', () => {
    showDesignCodeModal();
  });

function renderSpacecraftCanvas(telemetry) {
  const connected = getConnectedPartUids();
  const occupied = getOccupiedNodes();
  const drag = appState.activeDrag;
  const isDragging = drag && drag.mv;

  let html = appState.mission.parts.map(part => {
    const [w, h] = getPartSize(part);
    const selectedClass = part.u === appState.selectedPartUid ? ' sel' : '';
    const floatingClass = connected.has(part.u) ? '' : ' float';
    return `
      <img class="pt${selectedClass}${floatingClass}"
           tabindex="0"
           role="button"
           aria-label="${getPartDisplayName(part.id)}"
           data-u="${part.u}"
           src="assets/${getAssetFilename(part.id)}.svg"
           style="left:${part.x - w / 2}px; top:${part.y - h / 2}px; width:${w}px; height:${h}px; transform:rotate(${part.rot}deg) scaleX(${part.flip ? -1 : 1});"
           draggable="false">
    `;
  }).join('');

  const draggedSubtree = isDragging ? new Set(getSubtreeUids(drag.u)) : new Set();
  appState.mission.parts.forEach(part => {
    ATTACHMENT_NODES[getNodeCategory(part.id)].forEach(([nodeName]) => {
      if (occupied.has(`${part.u}:${nodeName}`)) return;
      const [x, y] = getNodeWorldPosition(part, nodeName);
      const nodeCls = isDragging ? (draggedSubtree.has(part.u) ? 'own' : '') : 'idle';
      html += `<i class="nd ${nodeCls}" style="left:${x}px; top:${y}px;"></i>`;
    });
  });

  if (isDragging && drag.sn) {
    const [snapX, snapY] = getNodeWorldPosition(findPartByUid(drag.sn.h), drag.sn.hn);
    html += `<i class="nd hot" style="left:${snapX}px; top:${snapY}px;"></i>`;
  }

  if (connected.size > 0) {
    html += `
      <i class="com" title="Center of mass" style="left:${telemetry.comx}px; top:${telemetry.comy}px;"></i>
      <u class="thr" title="Engine thrust line" style="left:${telemetry.engx}px; top:${telemetry.box[0]}px; height:${telemetry.box[1] - telemetry.box[0]}px;"></u>
    `;
  }

  return html;
}





function autoFitBuilderCanvas() {
  const cv = $('#cv');
  if (!cv) return;
  const idealZoom = Math.max(0.75, Math.min(1.05, +(cv.clientWidth / 920).toFixed(2)));
  appState.canvasZoom = idealZoom;

  cv.scrollLeft = Math.max(0, (960 * idealZoom - cv.clientWidth) / 2);
  cv.scrollTop = 40;
}

function renderFlightSpacecraft() {
  const connected = getConnectedPartUids();
  const parts = appState.mission.parts.filter(p => connected.has(p.u));
  if (!parts.length) return '';

  const bounds = parts.map(p => {
    const [w, h] = getPartSize(p);
    const [rw, rh] = p.rot % 180 ? [h, w] : [w, h];
    return [p.x - rw / 2, p.y - rh / 2, p.x + rw / 2, p.y + rh / 2];
  });

  const minX = Math.min(...bounds.map(b => b[0]));
  const minY = Math.min(...bounds.map(b => b[1]));
  const maxX = Math.max(...bounds.map(b => b[2]));
  const maxY = Math.max(...bounds.map(b => b[3]));

  const scale = Math.min(1.2, 480 / (maxX - minX), 420 / (maxY - minY));
  const offsetX = 260 - ((minX + maxX) / 2) * scale;
  const offsetY = 40 - minY * scale;

  const hullHtml = parts.map(p => {
    const [w, h] = getPartSize(p);
    return `<img class="pt" src="assets/${getAssetFilename(p.id)}.svg" alt="" style="left:${p.x * scale + offsetX - (w * scale) / 2}px; top:${p.y * scale + offsetY - (h * scale) / 2}px; width:${w * scale}px; height:${h * scale}px; transform:rotate(${p.rot}deg) scaleX(${p.flip ? -1 : 1});" draggable="false">`;
  }).join('');

  const plumeHtml = parts.filter(p => getNodeCategory(p.id) === 'eng').map(p => {
    const [, h] = getPartSize(p);
    return `<img class="pt plume" src="assets/${DATA.engines[p.id].pl}.svg" alt="" style="left:${p.x * scale + offsetX - 40 * scale}px; top:${p.y * scale + offsetY + (h * scale) / 2 - 14 * scale}px; width:${80 * scale}px; z-index:-1;" draggable="false">`;
  }).join('');

  return hullHtml + plumeHtml;
}

/** Turns one engine flight step into a log line. The engine decides; this only describes. */
function describeFlightStep(step) {
  if (step.key === 'event') {
    const text = CONTENT.eventText[step.values.eventId];
    return `<b>Event: ${text.n}.</b> ${text.txt} ${step.ok ? text.immT : text.hitT}`;
  }
  if (step.key === 'briefing') return buildPlanetBriefing(step.values.contractId, true);
  return fillTemplate(CONTENT.flightLog[step.key], step.values);
}

const getFlightStepClass = step => (!step.ok ? 'bad' : ['ascent', 'arrival', 'event'].includes(step.key) ? 'ok' : '');

export async function startLaunchSequence() {
  const mission = appState.mission;
  const telemetry = mission.d;
  const flight = resolveFlight(telemetry);
  mission.res = null;

  navigateToStage(3);
  appState.isFlying = true;

  $('#s3').innerHTML = `
    <div class="fl3">
      <div id="flv"><div class="flc" id="fl">${renderFlightSpacecraft()}</div></div>
      <div class="box" aria-live="polite">
        <h2>3. Launch and operate</h2>
        <div id="log"></div>
        <button type="button" class="go" id="fnext" hidden data-a="deb">See debrief &rarr;</button>
      </div>
    </div>
  `;
  $('#nav').querySelectorAll('button').forEach(btn => { btn.disabled = true; });

  const logEl = $('#log');
  const craftEl = $('#fl');

  const logMessage = async (html, statusClass = '') => {
    AudioEngine.tick();
    logEl.innerHTML += `<p class="${statusClass}">${html}</p>`;
    await delay(900);
  };

  void craftEl.offsetWidth;
  craftEl.classList.add('fly');
  AudioEngine.launch();

  for (const step of flight.steps) {
    if (step.key === 'arrival') {
      craftEl.style.transition = 'none';
      craftEl.classList.remove('fly');
      void craftEl.offsetWidth;
      craftEl.style.transition = '';
      craftEl.classList.add('orbit');
    }
    if (step.key === 'sun_dive') $('#flv').classList.add('sunhit');
    if (step.key === 'sun_boom') {
      craftEl.classList.add('burn');
      await delay(1500);
    }
    if (step.fatal) {
      craftEl.classList.add('boom');
      AudioEngine.fail();
    }
    await logMessage(describeFlightStep(step), getFlightStepClass(step));
  }

  concludeMission(flight);
}

/** Stores the engine's result, plays the end sound and unlocks the debrief. */
function concludeMission(flight) {
  const mission = appState.mission;
  const contract = mission.d.C;

  if (flight.outcome === 'success') AudioEngine.win();
  else if (flight.outcome === 'partial') AudioEngine.partial();

  appState.highScores[contract.id] = Math.max(appState.highScores[contract.id] || 0, flight.score);
  try {
    localStorage.tm = JSON.stringify(appState.highScores);
  } catch {}

  const connectedUids = getConnectedPartUids();
  const news = recordMission({
    out: flight.outcome,
    C: contract,
    d: mission.d,
    st: flight.stars,
    rtg: mission.parts.some(p => p.id === 'rtg' && connectedUids.has(p.u)),
    sci: flight.science,
    hard: mission.hard
  });

  mission.res = {
    out: flight.outcome,
    why: flight.reasons,
    sci: flight.science,
    st: flight.stars,
    score: flight.score,
    rel: flight.reliability,
    news
  };
  appState.isFlying = false;

  const levelUp = news.find(item => item.k === 'lv');
  if (levelUp) showToast(`Level up! Level ${levelUp.lv}: ${levelUp.title}. ${levelUp.planet} unlocked.`);
  updateCareerButton();

  $('#fnext').hidden = false;
  $('#fnext').onclick = () => navigateToStage(4);
  $('#nav').querySelectorAll('button').forEach(btn => {
    btn.disabled = btn.dataset.st === '3';
  });
}

export function renderDebriefStage() {

  const telemetry = mission.d;
  const contract = telemetry.C;

  const installedIds = new Set(
    mission.parts.filter(p => getConnectedPartUids().has(p.u)).map(p => p.id)
  );

  const resourceMarginsTable = `
    <p><b>Six-resource margin summary</b></p>
    <table class="rt">
      <tr><th>Resource</th><th>Used</th><th>Limit</th><th>Margin</th><th>Source</th></tr>
      ${telemetry.rows.map(r => `
        <tr>
          <td>${r.n}</td>
          <td>${formatNumber(r.use)} ${r.u}</td>
          <td>${formatNumber(r.cap)} ${r.u}</td>
          <td class="${r.mg < 0 ? 'r' : r.mg < appState.marginThreshold ? 'y' : 'g'}">${toPercent(r.mg)}%</td>
          <td><small>${r.src}</small></td>
        </tr>
      `).join('')}
    </table>
  `;

  const traceabilityTable = `
    <p><b>Science traceability matrix</b></p>
    <table class="rt">
      <tr><th>Objective</th><th>Instrument on board</th></tr>
      ${contract.obj.map(obj => {
        const matching = obj.ins.filter(id => installedIds.has(id));
        return `<tr><td>${obj.n}</td><td class="${matching.length ? 'g' : 'r'}">${matching.length ? matching.map(id => DATA.parts[id].n).join(', ') : 'Not covered'}</td></tr>`;
      }).join('')}
    </table>
  `;

  const installedArray = [...installedIds];
  const similarityScore = realMission =>
    realMission.ins.filter(id => installedArray.includes(id)).length * 2 + (realMission.oi === mission.oi ? 1 : 0);

  const comparableMissions = DATA.realMissions
    .filter(m => m.dest === contract.dest)
    .sort((a, b) => similarityScore(b) - similarityScore(a))
    .slice(0, 2);

  const heritageList = `
    <p><b>Comparable historical missions</b> ${renderDataBadge(false)}</p>
    <ul>${comparableMissions.map(m => `<li><b>${m.n}</b> (${m.yr}): ${CONTENT.realMissionText[m.n]}</li>`).join('')}</ul>
  `;

  const missingSignOffs = Object.keys(telemetry.sign).filter(role => !telemetry.sign[role]).join(', ') || 'all signed';

  $('#s4').innerHTML = `
    <div class="box">
      ${renderCustomEmblemSvg(out, st)}
      <h2>4. Post-Mission Report — ${'★'.repeat(st) + '☆'.repeat(3 - st)} (${score} pts)</h2>
      <p><b>Outcome:</b> ${CONTENT.outcomeLabels[out]}.</p>
      ${renderCareerNews(news)}
      <p><b>Why it went this way &amp; what to change next</b></p>
      <ul>${(why.length ? why : ['All six resource budgets maintained healthy engineering margins.']).map(reason => `<li>${reason}</li>`).join('')}</ul>
      <p><b>Star criteria:</b> 2&#9733; requires ${contract.target} science pts (achieved ${sci.toFixed(0)}). 3&#9733; also requires all four role sign-offs (${missingSignOffs}) and total cost under $${Math.round(contract.cap * DATA.scoringWeights.threeStarBudgetFraction)}M (spent $${telemetry.cost}M).</p>
      ${resourceMarginsTable}
      ${traceabilityTable}
      ${heritageList}
      <p><b>Destination briefing: ${contract.n}</b></p>
      ${buildPlanetBriefing(contract.id, true)}
      <div class="row" style="margin-top:12px;">
        <button type="button" data-a="go2">Edit and re-simulate</button>
        <button type="button" data-a="go1">New mission</button>
      </div>
    </div>
  `;
}

/* ---- Career mode: debrief news and the Career window ---- */

function renderCareerNews(news = []) {
  const career = getCareer();
  const next = getNextGate();
  const items = news.map(item => item.k === 'lv'
    ? `<b class="g">LEVEL UP! Level ${item.lv}: ${item.title}</b><br>Unlocked: ${item.planet}. The destination list is back in solar-system order.<br>`
    : `<b class="y">Achievement:</b> ${item.n} - ${item.t}<br>`
  ).join('');

  const nextLine = next
    ? `Next unlock: ${getPlanetName(next.id)} (Level ${next.lv}): ${describeGateNeed(next)}.`
    : 'Every destination is unlocked.';

  const tip = career.streak >= 2
    ? `<br><b class="y">Mission Control tip:</b> ${career.streak} losses in a row. Press Starter ship in the Build screen (it reaches Earth and the Moon safely; Mars and beyond need extra tanks and a bigger launcher), keep the Rookie 10% margin, and fix every red meter before you launch. Each red meter explains how to fix it.`
    : '';

  return `<div class="pin">${items}<b>Career:</b> Level ${getCareerLevel()} - ${getCareerRank()}. ${nextLine}${tip}</div>`;
}

## Career mode and all planets

Earth, the Moon and Mars start open. Safe arrivals unlock Venus, Jupiter, Saturn, Uranus, Neptune, Mercury and finally the Sun, which can never be reached (the launch plays out why). Progress and achievements are saved in the browser (`localStorage.apogeeCareer`).

- New destinations are appended after the original three contracts in `js/api/mockData.js`, so contract indices, saved candidates and share codes keep working. Values are tagged `SIMPLIFIED_FOR_GAMEPLAY` or `UNVERIFIED_MEMORY`, not official NASA data.
- Far from the Sun, solar wings give almost nothing: the radioisotope generator (`rtg`) makes power regardless of sunlight.
- Each destination can set its own design life (`contract.life`); the original three still use the 36-month default.
- Planet briefing text and the Sun dive text live in `js/content.js`.
